Q1. WAQ to find employees who earn more than then average salary of their department?
way1:
select first_name
from employees e1
where salary >(
select avg(salary)
from employees e2
where e1.department_id = e2.department_id
);

way2:
select first_name,salary
from employees e join 
(select department_id, avg(Salary) as sal
from employees 
group by department_id) d
on e.department_id = d.department_id
where e.salary > d.sal;

Q12. Find the category that has the highest total sales ?
SELECT top 1 c.category_id, c.category_name, sum(quantity * unit_price) total_sale
from products P join order_item O
on P.product_id = O.product_id
join category C
on O.category_id = C.category_id
order by total_sale desc;
